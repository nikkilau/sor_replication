\begin{frame}{Motivation: When FIFO Fails}
\begin{columns}[T]
\column{0.55\textwidth}
\begin{block}{Real-world problem}
\begin{itemize}
  \item Surgical endoscopy waiting lists
  \item Patients' urgency \textbf{changes} while waiting
  \item Observed service order $\neq$ arrival order
  \item Large ``overtakes'': bumped up \& bumped down
\end{itemize}
\end{block}

\column{0.45\textwidth}
\begin{block}{Research question}
\small
Can \textbf{stochastic priority switching} explain observed non-FIFO
behavior when the true scheduling rule is unknown?
\end{block}
\end{columns}

\vspace{1em}
\begin{center}
\textbf{Core idea:} Model an unknown service discipline as a random walk over priority classes.
\end{center}
\end{frame}

% ════════════════════════════════════════
% Slide 3: Model
% ════════════════════════════════════════
\begin{frame}{Queueing Model}
\begin{columns}[T]
\column{0.5\textwidth}
\textbf{M/M/c with $K$ priority classes}
\begin{itemize}
  \item $\lambda_k$: arrival rate of class $k$
  \item $\mu_k$: service rate of class $k$
  \item $c$: number of servers
  \item Lower index = higher priority
  \item Preemptive priority
\end{itemize}

\column{0.5\textwidth}
\textbf{Switching matrix $\Theta = (h_{ij})$}
\begin{itemize}
  \item $h_{i,i-1}$: upgrade rate
  \item $h_{i,i+1}$: downgrade rate
  \item $h_{ij}\;(|i-j|>1)$: skip-grade jumps
  \item Exponential switching times
  \item $\Rightarrow$ memoryless CTMC
\end{itemize}
\end{columns}

\vspace{1em}
\begin{center}
\small Customers randomly switch priority classes while waiting --- both \textbf{up} and \textbf{down}.
\end{center}
\end{frame}

% ════════════════════════════════════════
% Slide 4: Methods
% ════════════════════════════════════════
\begin{frame}{Methods: Dual Approach}

\begin{columns}[T]
\column{0.48\textwidth}
\begin{block}{1. Discrete-Event Simulation (Ciw)}
\begin{itemize}
  \item Poisson arrivals + Exp service + Exp switching
  \item Preemptive priority events
  \item Flexible: supports non-exponential distributions
  \item Noisy: requires many trials
\end{itemize}
\end{block}

\column{0.52\textwidth}
\begin{block}{2. Dual Markov Chain (analytical)}
\begin{itemize}
  \item \textbf{Ergodic chain:} state $s = (s_0,\ldots,s_{K-1})$
  \item $\to$ steady-state $L_k$, queue composition
  \item \textbf{Absorbing chain:} tagged customer
  \item $\to$ sojourn time $W$ (phase-type distribution)
\end{itemize}
\end{block}
\end{columns}

\vspace{1em}
\begin{alertblock}{Challenge: infinite state space}
Truncate with \textbf{finite bound $b$} $\;\to\;$ validate via $Q(b)$ and $P(b)$.
\end{alertblock}
\end{frame}
